// Genere une cle pour un acheteur
// Usage : node keygen.js lifetime  ou  node keygen.js weekly
const fs = require('fs');
const plan = process.argv[2] || 'weekly';
let days = 7;
if (plan === 'lifetime') days = 3650;
if (plan === 'monthly') days = 30;
if (plan === 'daily') days = 1;

function genKey() {
  const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789';
  let k = 'ZENO-';
  for (let i=0;i<8;i++) k += chars[Math.floor(Math.random()*chars.length)];
  return k;
}
const key = genKey();
const keys = JSON.parse(fs.readFileSync('keys.json','utf8'));
keys[key] = { hwid: null, expires: Math.floor(Date.now()/1000) + days*86400, plan, discordId: "" };
fs.writeFileSync('keys.json', JSON.stringify(keys, null, 2));
console.log('Nouvelle cle : ' + key + ' | plan : ' + plan + ' | expire dans ' + days + ' jours');
console.log('Envoie cette cle au client, il la met dans son script.');
